🧩 1. Fonction finale (calcul immédiat de la différence)
Cette fonction calcule exactement ce que tu veux :
👉 ce que la personne aurait dû payer jusqu’à maintenant
👉 ce qu’elle a réellement payé
👉 la différence à facturer immédiatement
function calculerDifferenceImmediate({
  moisEcoules,
  tarifAncien = 1.99,   // Premium en CAD
  tarifNouveau = 4.99   // Platine en CAD
}) {
  const montantDu = moisEcoules * tarifNouveau;   // Ce qu'il aurait dû payer
  const montantPaye = moisEcoules * tarifAncien;  // Ce qu'il a payé
  const difference = montantDu - montantPaye;     // À facturer maintenant

  return {
    montantDu: montantDu.toFixed(2),
    montantPaye: montantPaye.toFixed(2),
    differenceImmediate: difference.toFixed(2)
  };
}
